27 bool CNTV2Card::SubscribeEvent (const INTERRUPT_ENUMS id)
28 {
29  if (!IsOpen())
30  {DIFAIL("Cannot subscribe to '" << NTV2CfgInterrupt::IntName(id) << "' -- device not open"); return false;}
31  if (!NTV2_IS_VALID_INTERRUPT_ENUM(id))
32  {DIFAIL("Cannot subscribe to '" << NTV2CfgInterrupt::IntName(id) << "' -- bad interrupt ID"); return false;}
33 #if defined(NTV2_NUB_CLIENT_SUPPORT)
34  if (IsRemote())
35  { NTV2ConfigureInterrupt msg(this);
36  return _pRPCAPI->NTV2MessageRemote(msg.doSubscribe(id)) && msg.isSuccess();
37  }
38 #endif// defined(NTV2_NUB_CLIENT_SUPPORT)
39 #if defined(MSWindows)
40  return WinConfigureSubscription (true, id);
41 #else
42  DIDBG("Subscribe to '" << NTV2CfgInterrupt::IntName(id) << "' is a no-op on this platform");
43  return true; // Non-Windows platforms reply "good to go"
44 #endif
45 }

126  return false;
127  do

150 bool CNTV2Card::GetOutputFieldID (const NTV2Channel channel, NTV2FieldID & outFieldID)
151 {
152  // CHANNEL1 CHANNEL2 CHANNEL3 CHANNEL4 CHANNEL5 CHANNEL6 CHANNEL7 CHANNEL8
153  static ULWord regNum [] = {kRegStatus, kRegStatus, kRegStatus, kRegStatus, kRegStatus2, kRegStatus2, kRegStatus2, kRegStatus2, 0};
154  static ULWord bitShift [] = { 23, 5, 3, 1, 9, 7, 5, 3, 0};
155 
156  // Check status register to see if it is the one we want.

159  ? NTV2FieldID((statusValue >> bitShift[channel]) & 0x1)
